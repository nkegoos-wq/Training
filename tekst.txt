let data = JSON.parse(localStorage.getItem(STORAGE) || "{}");

let currentWorkout = "A";

let currentWeek = getMonday(new Date());

let exerciseDatabase = [];

function saveData() {

  localStorage.setItem(STORAGE, JSON.stringify(data));

}

function dateKey(date) {

  return date.toISOString().slice(0, 10);

}

function getMonday(date) {

  const d = new Date(date);

  d.setHours(0, 0, 0, 0);

  const day = (d.getDay() + 6) % 7;

  d.setDate(d.getDate() - day);

  return d;

}

function addDays(date, days) {

  const d = new Date(date);

  d.setDate(d.getDate() + days);

  return d;

}

function weekKey() {

  return dateKey(currentWeek);

}

function weekData() {

  const key = weekKey();

  if (!data[key]) data[key] = {};

  if (!data[key][currentWorkout]) {

    data[key][currentWorkout] = {};

  }

  return data[key][currentWorkout];

}

function getExerciseData(index) {

  const exercises = weekData();

  if (!exercises[index]) {

    exercises[index] = {

      sets: [

        { weight: "", reps: "" },

        { weight: "", reps: "" },

        { weight: "", reps: "" }

      ],

      notes: "",

      done: false

    };

  }

  return exercises[index];

}

function previousWeekData(index) {

  const previousMonday = addDays(currentWeek, -7);

  const key = dateKey(previousMonday);

  return data[key]?.[currentWorkout]?.[index] || null;

}

function bestWeight(index) {

  let best = 0;

  Object.values(data).forEach(week => {

    const exercise = week?.[currentWorkout]?.[index];

    if (!exercise || !exercise.sets) return;

    exercise.sets.forEach(set => {

      const weight = parseFloat(

        String(set.weight).replace(",", ".")

      );

      if (!isNaN(weight) && weight > best) {

        best = weight;

      }

    });

  });

  return best;

}

function escapeHtml(value) {

  return String(value ?? "").replace(/[&<>"']/g, char => ({

    "&": "&amp;",

    "<": "&lt;",

    ">": "&gt;",

    '"': "&quot;",

    "'": "&#039;"

  })[char]);

}

function updateWeekLabel() {

  const end = addDays(currentWeek, 6);

  const options = {

    day: "numeric",

    month: "short"

  };

  document.getElementById("weekLabel").textContent =

    currentWeek.toLocaleDateString("nl-BE", options) +

    " – " +

    end.toLocaleDateString("nl-BE", options);

  const thisMonday = getMonday(new Date());

  document.getElementById("nextWeek").disabled =

    currentWeek >= thisMonday;

}

function updateProgress() {

  const exercises = weekData();

  const total = workouts[currentWorkout].length;

  const completed = workouts[currentWorkout].filter(

    (_, index) => exercises[index]?.done

  ).length;

  document.getElementById("progressText").textContent =

    `${completed} / ${total} oefeningen`;

  document.getElementById("progressFill").style.width =

    `${(completed / total) * 100}%`;

}

function updateBestLabels() {

  workouts[currentWorkout].forEach((exercise, index) => {

    const element = document.getElementById(`best-${index}`);

    if (element) {

      const best = bestWeight(index);

      element.textContent = best > 0

        ? `Beste geregistreerde gewicht: ${best} kg`

        : "Nog geen beste gewicht geregistreerd";

    }

  });

}

function normalizeName(name) {

  return name.toLowerCase()

    .replace(/[^a-z0-9]/g, "");

}

function getExerciseImage(exercise) {

  if (!exerciseDatabase.length) {

    return `

      <div class="image-placeholder">

        Afbeelding wordt geladen...

      </div>

    `;

  }

  const searchName = normalizeName(exercise.search);

  let match = exerciseDatabase.find(item =>

    normalizeName(item.name) === searchName

  );

  if (!match) {

    match = exerciseDatabase.find(item =>

      normalizeName(item.name).includes(searchName) ||

      searchName.includes(normalizeName(item.name))

    );

  }

  if (!match || !match.images || !match.images.length) {

    return `

      <div class="image-placeholder">

        Geen afbeelding beschikbaar

      </div>

    `;

  }

  const imageUrl =

    "https://raw.githubusercontent.com/yuhonas/free-exercise-db/main/exercises/" +

    match.images[0];

  return `

    <img

      class="exercise-image"

      src="${escapeHtml(imageUrl)}"

      alt="${escapeHtml(exercise.name)}"

      loading="lazy"

      onerror="this.style.display='none'; this.nextElementSibling.style.display='flex';"

    >

    <div class="image-placeholder" style="display:none">

      Afbeelding niet beschikbaar

    </div>

  `;

}

async function loadExerciseDatabase() {

  try {

    const response = await fetch(

      "https://raw.githubusercontent.com/yuhonas/free-exercise-db/main/dist/exercises.json"

    );

    if (!response.ok) {

      throw new Error("Database niet bereikbaar");

    }

    exerciseDatabase = await response.json();

    render();

  } catch (error) {

    console.error("Afbeeldingen konden niet geladen worden:", error);

    exerciseDatabase = [];

    render();

  }

}

  const next = addDays(currentWeek, 7);

  const thisMonday = getMonday(new Date());

  if (next <= thisMonday) {

    currentWeek = next;

    render();

  }

});

// Alle oefeningen als voltooid markeren

document.getElementById("completeTraining").addEventListener("click", () => {

  workouts[currentWorkout].forEach((exercise, index) => {

    getExerciseData(index).done = true;

  });

  saveData();

  render();

});

// Huidige training wissen

document.getElementById("clearTraining").addEventListener("click", () => {

  const confirmed = confirm(

    "Wil je alle gegevens van deze training in deze week wissen?"

  );

  if (!confirmed) return;

  const key = weekKey();

  if (data[key]) {

    delete data[key][currentWorkout];

    if (Object.keys(data[key]).length === 0) {

      delete data[key];

    }

  }

  saveData();

  render();

});
